cmake_minimum_required(VERSION 3.24)
project(Burnside VERSION 1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Burnside requires Windows x64 and Visual Studio 2022.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_SUPPRESS_REGENERATION ON)
set(CMAKE_CONFIGURATION_TYPES "Debug;Release" CACHE STRING "" FORCE)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/Binaries")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/Binaries")
foreach(config Debug Release)
    string(TOUPPER "${config}" upper)
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${upper} "${CMAKE_SOURCE_DIR}/Binaries/${config}")
    set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY_${upper} "${CMAKE_SOURCE_DIR}/Binaries/${config}")
endforeach()
add_compile_options(/W4 /permissive- /utf-8 /MP /EHsc)
add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
add_library(BurnsideCore STATIC Src/Core/Cube.cpp Src/Core/Solver.cpp Src/Core/Timeline.cpp
    Src/Core/Cube.h Src/Core/Solver.h Src/Core/Timeline.h)
target_include_directories(BurnsideCore PUBLIC Src)
add_executable(BurnsideCLI Src/CLI/Main.cpp)
target_link_libraries(BurnsideCLI PRIVATE BurnsideCore)
add_executable(BurnsideTests Src/Tests/CoreTests.cpp Src/App/Scene.cpp)
target_link_libraries(BurnsideTests PRIVATE BurnsideCore)
add_executable(Burnside WIN32 Src/App/Main.cpp Src/App/Renderer.cpp Src/App/Scene.cpp
    Src/App/Feedback.cpp Src/App/Feedback.h
    Src/App/ImguiRenderDX11.cpp Src/ThirdParty/Imgui/imgui.cpp Src/App/Renderer.h
    Src/App/Scene.h Src/App/ImguiRenderDX11.h Src/ThirdParty/Imgui/imgui.h)
target_compile_definitions(Burnside PRIVATE IMGUI_STATIC UNICODE _UNICODE)
target_include_directories(Burnside PRIVATE Src/ThirdParty/Imgui)
target_link_libraries(Burnside PRIVATE BurnsideCore d3d11 dxgi d3dcompiler gdi32 user32 shell32 comdlg32)
set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT Burnside)
foreach(target BurnsideCore BurnsideCLI BurnsideTests Burnside)
    set_target_properties(${target} PROPERTIES
        COMPILE_PDB_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/Intermediate/$<CONFIG>/${target}"
        PDB_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/Intermediate/$<CONFIG>/${target}"
        VS_DEBUGGER_WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
    if(NOT target STREQUAL "BurnsideCore")
        target_link_options(${target} PRIVATE "/ILK:${CMAKE_SOURCE_DIR}/Intermediate/$<CONFIG>/${target}/${target}.ilk")
    endif()
endforeach()
enable_testing()
add_test(NAME Core COMMAND BurnsideTests)
add_test(NAME CLI_StateSearch COMMAND BurnsideCLI --order 3 --scramble 3 --seed 42 --state-only --depth 6)
add_test(NAME CLI_Order10 COMMAND BurnsideCLI --order 10 --scramble 30 --seed 2026)
set_tests_properties(Core CLI_StateSearch CLI_Order10 PROPERTIES TIMEOUT 180)
foreach(order RANGE 2 10)
    add_test(NAME GUI_Order${order} COMMAND Burnside --smoke --order ${order}
        --capture "${CMAKE_SOURCE_DIR}/Intermediate/Order${order}.bmp")
    set_tests_properties(GUI_Order${order} PROPERTIES TIMEOUT 240 RUN_SERIAL TRUE)
endforeach()
